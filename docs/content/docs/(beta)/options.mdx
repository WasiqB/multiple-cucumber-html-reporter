Multiple Cucumber HTML Reporter offers a wide range of configuration options to customize your test reports.

## Main Options

<TypeTable
  type={{
    jsonDir: {
      description: 'The directory containing all the generated JSON files from your test runs',
      type: 'string',
      required: true,
      typeDescription: 'string',
    },
    reportPath: {
      description: 'The directory in which the final report needs to be saved. Both jsonDir and reportPath are relative from where the script is started',
      type: 'string',
      required: true,
      typeDescription: 'string'
    },
    staticFilePath: {
      description: <>If <code>true</code>, each feature will get a static filename for the HTML. Use this feature only if you are not running multiple instances of the same tests</>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
  }}
/>

## Report Behavior Options

<TypeTable
  type={{
    openReportInBrowser: {
      description: <>If <code>true</code>, the report will automatically be opened in the default browser of the operating system</>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
    saveCollectedJSON: {
      description: <>
        This module will first merge all the JSON files to a single file and then enrich it with data that is used for the report. Setting this to <code>true</code> saves the merged JSON and the enriched JSON in the <code>reportPath</code>.
      </>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
    disableLog: {
      description: 'Disables the summary report log in the console.',
      type: 'boolean',
      deprecated: true,
      default: false,
      typeDescription: 'boolean'
    },
    externalizeMedia: {
      description: <>
        If <code>true</code>, image and video attachments are written to separate files under <code>{'<reportPath>/assets/media/'}</code> instead of being inlined as <code>data:</code> URIs in the feature page's HTML. Keeps feature pages small and lets the browser load each screenshot or video lazily instead of parsing everything up front.

        <Callout title="Note" type="info">
          Only image and video attachments are affected. Text, JSON and HTML log attachments are always inlined.
        </Callout>
      </>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
    logging: {
      description: <>
        Controls reporter logging.

        LogLevel: <code>silent</code>, <code>error</code>, <code>warn</code>, <code>info</code>, <code>debug</code>, <code>trace</code>
      </>,
      type: 'boolean | LogLevel | { level: LogLevel; enabled: boolean }',
      default: 'info',
      typeDescription: 'boolean | LogLevel | { level: LogLevel; enabled: boolean }'
    }
  }}
/>

<TypeTable

    metadata: {

          Use metadata of <code>Record&lt;string, Metadata&gt;</code> type for feature specific metadata or <code>Metadata</code> for common metadata across run.
        </Callout>
      </>,
      type: <>
        <code>Metadata</code> | {' '}

    customData: {

      typeDescription: 'CustomData',

## Duration Options

<TypeTable
  type={{
    displayDuration: {
      description: (
        <>
          If set to <code>true</code>, the duration of steps, scenarios, and features is displayed on the Features overview and single feature page in an easily readable format.
          <Callout title="Note" type="info">
            CucumberJS 2 and 3 use milliseconds, while others use nanoseconds. You may need to use <code>durationInMS: true</code> if your durations are not showing correctly.
          </Callout>
        </>
      ),
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
    durationInMS: {
      description: <>
        If set to <code>true</code>, step durations are expected to be in <strong>milliseconds</strong>. This parameter relies on <code>displayDuration: true</code>.
      </>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
    durationAggregation: {
      description: (
        <>
          Controls how feature duration is calculated:
          <ul>
            <li><code>sum</code>: sum all scenario durations in a feature (default)</li>
            <li><code>wallClock</code>: use elapsed wall-clock time (<code>latest scenario end - earliest scenario start</code>). Each timed scenario must provide <code>start_timestamp</code>.</li>
          </ul>
        </>
      ),
      type: 'sum | wallClock',
      default: 'sum',
      typeDescription: 'sum | wallClock'
    },
    displayReportTime: {
      description: <>If set to <code>true</code>, the report will display the time of the feature.</>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    }
  }}
/>

## Attachment Display Options

<TypeTable
  type={{
    attachmentLayout: {
      description: (
        <>
          Controls how step attachments (text, JSON, HTML, image, video, other) are presented on feature pages.
          <ul>
            <li><code>modal</code>: attachments are listed under each step; clicking one opens it in a popup (see the <code>modal*</code> options below).</li>
            <li><code>inline</code>: classic pre-4.x style. One link per attachment type ("+ Show Info", "+ Screenshot", etc.) next to the step text; each expands its content directly below the step.</li>
          </ul>
        </>
      ),
      type: 'modal | inline',
      default: 'modal',
      typeDescription: 'modal | inline'
    },
    modalBackdrop: {
      description: <>If <code>true</code>, the attachment popup (<code>attachmentLayout: 'modal'</code>) shows a dark backdrop, blocks interaction with the rest of the page, and closes on click-outside.</>,
      type: 'boolean',
      default: true,
      typeDescription: 'boolean'
    },
    modalDraggable: {
      description: <>If <code>true</code>, the attachment popup can be repositioned by dragging its header.</>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
    modalResizable: {
      description: <>If <code>true</code>, the attachment popup can be resized from its edges and corners.</>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
    modalShowContext: {
      description: <>If <code>true</code>, the attachment popup header shows the step text (or, for a detected video, the scenario text) as context.</>,
      type: 'boolean',
      default: false,
      typeDescription: 'boolean'
    },
  }}
/>
